<script>
let resultado = localStorage.getItem("resultado");

// ✅ Validación
if (!resultado) {
    document.getElementById("resultadoTexto").innerText = "No hay resultados disponibles.";
} else {

    resultado = parseInt(resultado);

    let texto = "";

    if (resultado <= 4) {
        texto = "Perfil bajo en áreas analizadas";
    } else if (resultado <= 7) {
        texto = "Intereses variados";
    } else {
        texto = "Alta afinidad con tecnología o ciencias";
    }

    document.getElementById("resultadoTexto").innerText = texto;

    // 📊 Gráfica mejorada
    let ctx = document.getElementById('grafica').getContext('2d');

    new Chart(ctx, {
        type: 'bar',
        data: {
            labels: ['Tu puntaje'],
            datasets: [{
                label: 'Resultado del test',
                data: [resultado]
            }]
        },
        options: {
            scales: {
                y: {
                    beginAtZero: true,
                    max: 10
                }
            }
        }
    });
}
</script>
